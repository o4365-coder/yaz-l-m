class BankaHesabi:
   def __init__(self,sahip,bakiye):
       self.sahip=sahip
       self.bakiye=bakiye
        
        
   def para_yatir(self,miktar):
       self.bakiye=self.bakiye+miktar
       print(miktar,"TL yatırıldı. Yeni bakiye",self.bakiye)
        
   def para_cek(self,miktar):
       if miktar>self.bakiye:
           print("Yetersiz Bakiye")
       else:
           self.bakiye=self.bakiye-miktar
           print(miktar,"TL çekildi,Kalan:",self.bakiye)
               

hesap=BankaHesabi("Ahmet",1000)
hesap.para_yatir(500)
hesap.para_yatir(500)
hesap.para_yatir(750)
hesap.para_cek(2750)
